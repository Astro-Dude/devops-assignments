$ kubectl -n ingress-nginx logs deploy/ingress-nginx-controller --since=1m | grep 'broken.*api/stats' | tail -1 | cut -c1-220

## ROOT CAUSE: the /api rule targets port 8080 on tickethub-backend; that Service only exposes 8000, so the
## controller builds an upstream with no endpoints and answers 503 itself.
## FIX: reference the Service port by NAME so a port renumbering can never break the Ingress again
$ kubectl -n tickethub-broken patch ingress tickethub --type=json -p '[{"op":"replace","path":"/spec/rules/0/http/paths/0/backend/service/port","value":{"name":"http"}}]'
ingress.networking.k8s.io/tickethub patched
