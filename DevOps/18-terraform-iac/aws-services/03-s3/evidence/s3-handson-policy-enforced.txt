# The DenyInsecureTransport statement is being enforced: the emulator endpoint is plain http://, so aws:SecureTransport=false
$ aws s3api get-object --bucket astro-dude-s18-research --key reports/2026/q3.txt /dev/stdout

# remove the policy (in real AWS the account root user can always delete a bucket policy that locks everyone out)
$ aws s3api delete-bucket-policy --bucket astro-dude-s18-research

$ aws s3api get-object --bucket astro-dude-s18-research --key reports/2026/q3.txt /dev/stdout --query ContentLength

# cleanup (versioned bucket: delete every version, then the bucket)

# 'rb --force' only deletes CURRENT versions; noncurrent versions and delete markers still count as content
$ aws s3api list-object-versions --bucket astro-dude-s18-research --query {Versions:length(Versions || `[]`),DeleteMarkers:length(DeleteMarkers || `[]`)}
{
    "Versions": 5,
    "DeleteMarkers": 4
}

$ (loop) aws s3api delete-object --bucket astro-dude-s18-research --key <key> --version-id <id>   # for every version and delete marker
