# EBS: an extra 10 GiB data volume, attached as /dev/sdf, then snapshotted
$ aws ec2 create-volume --availability-zone ap-south-1a --size 10 --volume-type gp3 --query VolumeId --output text
vol-63575e78aadc1862f

$ aws ec2 attach-volume --volume-id vol-63575e78aadc1862f --instance-id i-375f4ae752b18b194 --device /dev/sdf --query {Volume:VolumeId,Device:Device,State:State}
{
    "Volume": "vol-63575e78aadc1862f",
    "Device": "/dev/sdf",
    "State": "attaching"
}

$ aws ec2 create-snapshot --volume-id vol-63575e78aadc1862f --description s18 data backup --query {SnapshotId:SnapshotId,State:State,VolumeSize:VolumeSize}
{
    "SnapshotId": "snap-38dcda31591f804eb",
    "State": "completed",
    "VolumeSize": 10
}

}
