########## THE MISTAKE: COMMIT A SECRET, THEN 'REMOVE' IT ##########
$ git init -q leaky-repo

$ kubectl create secret generic db-secret --from-literal=DB_PASSWORD='Pr0d-P@ssw0rd' --dry-run=client -o yaml > db-secret.yaml

$ cat db-secret.yaml
apiVersion: v1
data:
  DB_PASSWORD: UHIwZC1QQHNzdzByZA==
kind: Secret
metadata:
  name: db-secret

$ git add db-secret.yaml && git commit -q -m 'add db secret' && git log --oneline
9d97621 add db secret

$ git rm -q db-secret.yaml && git commit -q -m 'oops, remove secret' && git log --oneline
ef9bc02 oops, remove secret
9d97621 add db secret

$ ls

# the file is gone from the working tree, but every clone still has it in history:
$ git log --oneline --all -- db-secret.yaml
ef9bc02 oops, remove secret
9d97621 add db secret

$ git show HEAD~1:db-secret.yaml | grep DB_PASSWORD
  DB_PASSWORD: UHIwZC1QQHNzdzByZA==

$ git show HEAD~1:db-secret.yaml | awk '/DB_PASSWORD/{print $2}' | base64 -d; echo
Pr0d-P@ssw0rd

########## THE SAFE WORKFLOW: VALUES OUTSIDE GIT ##########
$ git init -q safe-repo

$ printf 'DB_USER=app\nDB_PASSWORD=Pr0d-P@ssw0rd\n' > .env

$ printf '.env\n*.secret.yaml\n' > .gitignore

$ git check-ignore -v .env
.gitignore:1:.env	.env

$ git status --short --untracked-files=all
?? .gitignore

$ git add . && git commit -q -m 'app config, no secrets' && git ls-files
.gitignore

# render the Secret from the ignored file (nothing written to disk):
$ kubectl create secret generic db-secret --from-env-file=.env --dry-run=client -o yaml
apiVersion: v1
data:
  DB_PASSWORD: UHIwZC1QQHNzdzByZA==
  DB_USER: YXBw
kind: Secret
metadata:
  name: db-secret

# or create it straight in the cluster:
$ kubectl create namespace s12-secrets
namespace/s12-secrets created

$ kubectl -n s12-secrets create secret generic db-secret --from-env-file=.env
secret/db-secret created

$ kubectl -n s12-secrets get secret db-secret
NAME        TYPE     DATA   AGE
db-secret   Opaque   2      0s

$ kubectl -n s12-secrets get secret db-secret -o jsonpath='{.data.DB_PASSWORD}' | base64 -d; echo
Pr0d-P@ssw0rd

$ git log -p | grep -c 'Pr0d' || true
0

$ kubectl delete namespace s12-secrets
namespace "s12-secrets" deleted
